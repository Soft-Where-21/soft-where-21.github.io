import React, {useEffect, useMemo, useState} from 'react';

import toolsData from './tools.json';

import TimestampTool from './timestamp';
import JsonTool from './json';
import ColorPickerTool from './color-picker';
import RegexBuilderTool from './regex-builder';
import GpaTool from './gpa';
import PostgradTool from './postgrad';
import ComprehensiveTool from './comprehensive';

export function ToolPage() {
  const gradeExtensionRepoUrl = 'https://github.com/Soft-Where-21/byxt-grade-extension';
  const gradeExtensionDownloadUrl =

  const tools = Array.isArray(toolsData?.tools) ? toolsData.tools : [];
  const toolById = useMemo(() => new Map(tools.map((t) => [t.id, t])), [tools]);

  const toolComponents = useMemo(
    () => ({
      timestamp: <TimestampTool />,
      json: <JsonTool />,
      'color-picker': <ColorPickerTool />,
      'regex-builder': <RegexBuilderTool />,
      gpa: <GpaTool />,
      postgrad: <PostgradTool />,
      comprehensive: <ComprehensiveTool />,
    }),
    []
  );

  const [activeId, setActiveId] = useState(defaultId);

  // 支持 hash 直达：/tool#json
  useEffect(() => {
    if (typeof window === 'undefined') return;
    const hash = decodeURIComponent(window.location.hash || '').replace(/^#/, '').trim();
    if (hash && toolById.has(hash)) setActiveId(hash);
  }, [toolById]);

  useEffect(() => {
    if (typeof window === 'undefined') return;
    if (!activeId) return;
    window.history.replaceState(null, '', `#${encodeURIComponent(activeId)}`);
  }, [activeId]);

    <div className="toolPageShell">
      <div className="toolLayout">
        {/* Sidebar */}
        <aside className="toolSidebar">
          <div className="toolSidebarMeta">
            <span>工具</span>
            <small>{tools.length}</small>
          </div>
          <div className="toolSidebarList">
            {tools.map((t) => {
              const isActive = t.id === activeId;
              return (
                <button
                  key={t.id}
                  onClick={() => setActiveId(t.id)}
                  className={`toolItem ${isActive ? 'isActive' : ''}`}
                  type="button"
                  aria-label={`${t.name}：${t.tagline}`}
                >
                  <span className="toolItemShort" aria-hidden="true">
                    {t.shortName || t.name}
                  </span>
                  <span className="toolItemCopy">
                    <span className="toolItemName">{t.name}</span>
                    <span className="toolItemTagline">{t.tagline}</span>
                  </span>
                </button>
              );
            })}
          </div>
        </aside>

        {/* Fixed Renderer */}
        <section className="toolMain">
          <div className="card toolStage">
            <div className="toolStageHeader">
              <div className="toolStageTitle">
                <strong>{activeTool?.name || '工具'}</strong>
                {activeTool?.tagline && <span>{activeTool.tagline}</span>}
              </div>
              {activeId === 'postgrad' && (
                <div className="gradeExtensionActions" aria-label="成绩导入插件">
                  <span className="gradeExtensionIdentity">
                    <span className="gradeExtensionDot" aria-hidden="true" />
                    成绩导入插件
                  </span>
                  <a
                    className="gradeExtensionDownload"
                    href={gradeExtensionDownloadUrl}
                    aria-label="下载成绩导入插件 v0.2.0"
                  >
                    <span>下载</span>
                    <span className="gradeExtensionVersion">v0.2.0</span>
                  </a>
                  <a
                    className="gradeExtensionRepo"
                    href={gradeExtensionRepoUrl}
                    target="_blank"
                    rel="noopener noreferrer"
                  >
                    安装说明
                    <span aria-hidden="true">↗</span>
                  </a>
                </div>
              )}
            </div>
            <div className="toolStageBody">
              {toolComponents[activeId] || (
                <div style={{padding: 18, opacity: 0.75}}>
                  暂无该工具组件（占位）。
                </div>
              )}
            </div>
          </div>
        </section>
