import Link from '@docusaurus/Link';
import Heading from '@theme/Heading';
import repoData from './repo.json';

export function RepoList() {
  const org = repoData?.org ?? {};
  const repos = Array.isArray(repoData?.repos) ? [...repoData.repos] : [];

  function safeTime(value) {
    const t = new Date(value ?? 0).getTime();
    return Number.isFinite(t) ? t : 0;
  }

  function formatDate(value) {
    if (!value) return null;
    const d = new Date(value);
    if (Number.isNaN(d.getTime())) return String(value);
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${y}-${m}-${day}`;
  }

  function repoHref(repo) {
    if (repo?.url) return repo.url;
    if (org?.githubUrl && repo?.name) return `${org.githubUrl}/${repo.name}`;
    return '#';
  }

  function authorAvatar(author) {
    if (!author?.github) return null;
    const username = author.github.replace(/\/$/, '').split('/').pop();
    return `https://github.com/${username}.png?size=40`;
  }

  repos.sort((a, b) => {
    const fa = a?.featured ? 1 : 0;
    const fb = b?.featured ? 1 : 0;
    if (fa !== fb) return fb - fa;
    const ta = safeTime(a?.updatedAt);
    const tb = safeTime(b?.updatedAt);
    if (ta !== tb) return tb - ta;
    return String(a?.name ?? '').localeCompare(String(b?.name ?? ''), 'en');
  });

  const langColors = {
    'JavaScript': '#f1e05a',
    'TypeScript': '#3178c6',
    'Python': '#3572a5',
    'Java': '#b07219',
    'C++': '#f34b7d',
    'C': '#555555',
    'Rust': '#dea584',
    'Go': '#00add8',
    'HTML': '#e34c26',
    'CSS': '#563d7c',
    'SCSS': '#c6538c',
    'Shell': '#89e051',
    'Ruby': '#701516',
    'Jupyter Notebook': '#da5b0b',
  };

  return (
    <div className="container margin-vert--lg">
      {/* Header */}
      <div style={{
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'space-between',
        flexWrap: 'wrap',
        gap: 16,
        marginBottom: 32,
      }}>
        <div>
          <Heading as="h1" style={{marginBottom: 4}}>
            仓库
          </Heading>
          <p style={{margin: 0, fontSize: '1rem', opacity: 0.75}}>
            {org?.name ?? 'Soft-Where-21'} · 共 {repos.length} 个公开仓库
          </p>
        </div>
        {org?.repositoriesUrl ? (
          <Link
            className="button button--primary"
            href={org.repositoriesUrl}
            target="_blank"
            rel="noreferrer"
          >
            在 GitHub 查看全部
          </Link>
        ) : null}
      </div>

      {/* Cards */}
      <div className="row repoGrid">
        {repos.map((repo) => {
          const updated = formatDate(repo?.updatedAt);
          const href = repoHref(repo);
          const author = repo?.author;
          const avatar = authorAvatar(author);
          const dotColor = langColors[repo?.language] ?? 'var(--ifm-color-emphasis-500)';

          return (
            <div key={repo?.name ?? href} className="col col--4 margin-bottom--md">
              <div className="card repoCard">
                <div className="card__body repoCardBody">
                  <div className="repoCardHead">
                    <Heading as="h3" className="repoCardTitle">
                      <Link
                        href={href}
                        target="_blank"
                        rel="noreferrer"
                        className="repoCardLink"
                      >
                        {repo?.name ?? 'Unnamed Repo'}
                      </Link>
                    </Heading>
                    <div className="repoCardBadges">
                      {repo?.featured ? <span className="badge badge--primary">推荐</span> : null}
                      {repo?.archived ? <span className="badge badge--warning">Archived</span> : null}
                    </div>
                  </div>

                  <p className="repoCardDesc">
                    {repo?.description || '（暂无简介）'}
                  </p>

                  {(repo?.language || repo?.license) ? (
                    <div className="repoCardMeta">
                      {repo?.language ? (
                        <span className="repoLang">
                          <span className="repoLangDot" style={{backgroundColor: dotColor}} />
                          {repo.language}
                        </span>
                      ) : null}
                      {repo?.license ? <span>{repo.license}</span> : null}
                    </div>
                  ) : null}

                  <div className="repoCardFoot">
                    {author?.name ? (
                      <Link
                        href={author.github || '#'}
                        target="_blank"
                        rel="noreferrer"
                        className="repoAuthor"
                      >
                        {avatar ? (
                          <img
                            src={avatar}
                            alt={author.name}
                            width="18"
                            height="18"
                            className="repoAvatar"
                            loading="lazy"
                          />
                        ) : null}
                        <span>{author.name}</span>
                      </Link>
                    ) : <span />}

                    {updated ? (
                      <span className="repoUpdated">
                        {updated}
                      </span>
                    ) : null}
                  </div>
                </div>
              </div>
            </div>
          );
        })}
      </div>

      <style>{`
        .repoGrid { margin-left: -8px; margin-right: -8px; }
        .repoGrid > .col { padding-left: 8px; padding-right: 8px; }
        .repoCard {
          height: 100%;
          border: 1px solid var(--ifm-color-emphasis-200);
          border-radius: 10px;
          box-shadow: none;
          background: var(--ifm-card-background-color);
          transition: border-color .15s, box-shadow .15s;
        }
        .repoCard:hover {
          border-color: var(--ifm-color-primary);
          box-shadow: 0 4px 14px rgba(0,0,0,.07);
        }
        .repoCardBody {
          display: flex;
          flex-direction: column;
          gap: 8px;
          padding: 14px 16px 12px;
        }
        .repoCardHead {
          display: flex;
          align-items: flex-start;
          justify-content: space-between;
          gap: 8px;
        }
        .repoCardTitle {
          margin: 0;
          font-size: 0.95rem;
          line-height: 1.35;
          min-width: 0;
        }
        .repoCardLink {
          color: var(--ifm-color-primary);
          text-decoration: none;
          font-weight: 650;
          display: -webkit-box;
          -webkit-line-clamp: 2;
          -webkit-box-orient: vertical;
          overflow: hidden;
          word-break: break-word;
        }
        .repoCardBadges { display: flex; gap: 4px; flex-shrink: 0; }
        .repoCardDesc {
          margin: 0;
          line-height: 1.5;
          font-size: 0.82rem;
          opacity: 0.82;
          display: -webkit-box;
          -webkit-line-clamp: 3;
          -webkit-box-orient: vertical;
          overflow: hidden;
          min-height: 3.7em;
        }
        .repoCardMeta {
          display: flex;
          gap: 10px;
          align-items: center;
          font-size: 0.75rem;
          opacity: 0.65;
        }
        .repoLang { display: inline-flex; align-items: center; gap: 4px; }
        .repoLangDot {
          width: 8px;
          height: 8px;
          border-radius: 50%;
          display: inline-block;
        }
        .repoCardFoot {
          margin-top: auto;
          display: flex;
          align-items: center;
          justify-content: space-between;
          gap: 8px;
          padding-top: 8px;
          border-top: 1px solid var(--ifm-color-emphasis-200);
          font-size: 0.75rem;
        }
        .repoAuthor {
          display: inline-flex;
          align-items: center;
          gap: 5px;
          min-width: 0;
          text-decoration: none;
          color: var(--ifm-color-primary);
        }
        .repoAuthor span {
          overflow: hidden;
          text-overflow: ellipsis;
          white-space: nowrap;
        }
        .repoAvatar { border-radius: 50%; flex-shrink: 0; }
        .repoUpdated { opacity: 0.5; flex-shrink: 0; }
        @media (max-width: 996px) {
          .repoGrid .col--4 { flex: 0 0 50%; max-width: 50%; }
        }
        @media (max-width: 640px) {
          .repoGrid .col--4 { flex: 0 0 100%; max-width: 100%; }
        }
      `}</style>
    </div>
  );
}

<RepoList />
