import React, {useState} from 'react';
import Heading from '@theme/Heading';
import data2526 from './2025-2026/member.json';

{/* 新增届次只需：1) 加 JSON  2) 在 allTerms 数组中追加一项 */}

export function MemberPage() {
  const allTerms = [
    { id: '2025-2026', label: '2025 — 2026', data: data2526 },
  ];

  const [activeTermId, setActiveTermId] = useState(allTerms[0].id);
  const activeTerm = allTerms.find(t => t.id === activeTermId) || allTerms[0];
  const departments = Array.isArray(activeTerm.data?.departments) ? activeTerm.data.departments : [];
  const totalMembers = departments.reduce(
    (total, department) => total + (Array.isArray(department?.members) ? department.members.length : 0),
    0
  );

  const roleOrder = { '主席': 0, '部长': 1, '副部长': 2, '干事': 3 };

  function sortMembers(members) {
    return [...members].sort(
      (a, b) => (roleOrder[a?.role] ?? 99) - (roleOrder[b?.role] ?? 99)
    );
  }

  function getInitial(name) {
    return (name || '?').charAt(0);
  }

  function roleStyle(role) {
    if (role === '主席') {
      return { color: '#b37a1a', fontWeight: 600 };
    }
    if (role === '部长') {
      return { color: '#2563eb', fontWeight: 600 };
    }
    if (role === '副部长') {
      return { color: '#0891b2', fontWeight: 600 };
    }
    return { opacity: 0.5 };
  }

  return (
    <div className="memberPageShell">
      <header className="memberPageHeader">
        <Heading as="h1">学生会成员</Heading>
      </header>

      <div className="memberLayout">
        <aside className="memberTermPanel">
          <div className="memberTermSection">
            <div className="memberNavLabel">届次</div>
            <div className="memberTermList" role="group" aria-label="选择届次">
              {allTerms.map((term) => {
                const isActive = term.id === activeTermId;
                return (
                  <button
                    key={term.id}
                    type="button"
                    className={`memberTermButton ${isActive ? 'isActive' : ''}`}
                    aria-pressed={isActive}
                    onClick={() => setActiveTermId(term.id)}
                  >
                    {term.label}
                  </button>
                );
              })}
            </div>
          </div>

          <div className="memberDepartmentDirectory">
            <div className="memberDirectoryHeader">
              <span className="memberNavLabel">部门</span>
              <span>{departments.length}</span>
            </div>
            <nav className="memberDepartmentLinks" aria-label="部门目录">
              {departments.map((department, departmentIndex) => {
                const departmentSize = Array.isArray(department?.members)
                  ? department.members.length
                  : 0;
                return (
                  <a
                    key={department?.name}
                    href={`#member-department-${departmentIndex}`}
                    className="memberDepartmentLink"
                    style={{'--member-department-color': department?.color || 'var(--ifm-color-primary)'}}
                  >
                    <span className="memberDepartmentMarker" aria-hidden="true" />
                    <span className="memberDepartmentName">{department?.name}</span>
                    <span className="memberDepartmentCount">{departmentSize}</span>
                  </a>
                );
              })}
            </nav>
          </div>

          <div className="memberTermStats">
            共 {totalMembers} 人
          </div>
        </aside>

        <main className="memberDepartments">
          {departments.map((dept, departmentIndex) => {
            const members = sortMembers(dept?.members || []);

            return (
              <section
                key={dept?.name}
                id={`member-department-${departmentIndex}`}
                className="memberDepartment"
                style={{'--member-department-color': dept?.color || 'var(--ifm-color-primary)'}}
              >
                <div className="memberDepartmentHeader">
                  <Heading as="h2">{dept?.name}</Heading>
                  <span>{members.length} 人</span>
                </div>

                <div className="memberGrid">
                  {members.map((m, idx) => {
                    const hasAvatar = m?.avatar && m.avatar.trim() !== '';
                    const rs = roleStyle(m?.role);
                    const sid = (m?.studentId ?? '').toString().trim();

                    return (
                      <article
                        key={`${dept?.name}-${m?.name}-${idx}`}
                        className="memberCard"
                      >
                        {hasAvatar ? (
                          <div className="memberPhoto">
                            <img src={m.avatar} alt={m.name} loading="lazy" />
                          </div>
                        ) : (
                          <div className="memberPhoto memberPhotoFallback" aria-hidden="true">
                            <span>{getInitial(m?.name)}</span>
                          </div>
                        )}

                        <div className="memberInfo">
                          <div className="memberNameRow">
                            <div className="memberName">{m?.name ?? '未知'}</div>
                            <div className="memberRole" style={rs}>{m?.role || '成员'}</div>
                          </div>
                          {sid ? <div className="memberStudentId">{sid}</div> : null}
                        </div>
                      </article>
                    );
                  })}
                </div>
              </section>
            );
          })}
        </main>
      </div>
