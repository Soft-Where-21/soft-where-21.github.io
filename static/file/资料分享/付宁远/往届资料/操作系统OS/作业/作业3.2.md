public class OPT {

    private int capacity;
    private List<Integer> list = new ArrayList<Integer>();
    private int cnt = 0;
    public OPT(int[] arr,int capacity) {
        this.capacity = capacity;
        for (int i = 0; i < arr.length; i++) {
            if (list.size() < capacity) {
                if (!list.contains(arr[i])){
                    list.add(arr[i]);
                    cnt++;
                }
            } else {
                if(list.contains(arr[i]))
                    continue;
                else{
                    cnt++;
                    int change_index = 0;
                    int max_dis = 0;
                    for(int index = 0; index < capacity; index++){
                        int page = list.get(index);
                        int j;
                        for(j = i+1; j < arr.length; j++)
                            if(arr[j] == page)
                                break;
                        if(j-i > max_dis){
                            max_dis = j-i;
                            change_index = index;
                        }
                    }
                    list.set(change_index,arr[i]);
                }
            }
        }
        System.out.println(capacity + "个页框缺页" + cnt + "次");
    }
}

public class FIFO {

    private int capacity;
    private int index = 0;
    private Queue<Integer> queue ;
    private int cnt = 0;
    public FIFO(int[] arr,int capacity) {
        queue = new ArrayBlockingQueue<Integer>(capacity);
        this.capacity = capacity;
        for (int i = 0; i < arr.length; i++) {
            if (queue.size() < capacity) {
                if (!queue.contains(arr[i])) {
                    queue.add(arr[i]);
                } else {
                    continue;
                }
            } else {
                if (!queue.contains(arr[i])) {
                    queue.poll();
                    queue.add(arr[i]);
                } else {
                    continue;
                }
            }
            cnt++;
        }
        System.out.println(capacity + "个页框缺页" + cnt + "次");
    }
}

public class LRU {

    // 初始化一个容量为 5 的栈
    private MyStack<Integer> stack ;
    private int index = 0;// 一个数组下标的标识，用来除去栈底元素
    private int capacity;
    public LRU(int[] arr,int capacity) {
        this.capacity = capacity;
        stack = new MyStack<Integer>(capacity);
        for (int i = 0; i < arr.length; i++) {// 将数组里 数据存入栈中
            if (stack.getLength() < stack.size()) { // 判断栈不满
                if (!stack.exist(arr[i])) { // 判断栈中有没有该元素，有的话返回true
                    stack.push(arr[i]);
                    stack.traverse();
                } else { // 如果有该元素，删除掉栈中该元素，并将该元素置于栈顶
                    stack.removeSameElement(arr[i]);
                    stack.push(arr[i]);
                }
            } else { // 栈已满，删除到栈底元素
                if (stack.exist(arr[i])) { // 继续判断如果栈中存在相同元素，将该元素从栈中取出放到栈顶
                    stack.removeSameElement(arr[i]);
                    stack.push(arr[i]);
                } else { // 栈中没相同元素，删除栈底元素，在栈顶加新元素
                    stack.remove(arr[index++]);
                    stack.push(arr[i]);
                    stack.traverse(); // 遍历栈中元素
                }
            }
        }
        System.out.println(capacity + "个页框缺页" + stack.getNum() + "次");
    }
}
