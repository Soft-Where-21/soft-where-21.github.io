## 题目分析

​	见出题人知题意（×）

​	一道比较明显的贪心，每一个节目对后续观看节目有没有影响的参数是它结束的时间，那就把节目按照结束时间从小到大排序，然后从头开始，能选则选就可以使得最终观看节目数量最大

## 代码



```c++
#include<queue>
#include<stdio.h>
#include<iostream>
#include<algorithm>
#include<cmath>
#include<map>
#include<string>
#include<string.h>
#include<ctype.h>
#include<stdlib.h>
#define rep(i,st,en) for(int i=st;i<=en;i++)
typedef long long ll;
using namespace std;

inline int read()
{
    int x=0,f=1;
    char ch=getchar();
    while(ch<'0'||ch>'9'){if(ch=='-')f=-1;ch=getchar();}
    while(ch>='0'&&ch<='9'){x=x*10+ch-'0';ch=getchar();}
    return x*f;
}
int n,ans;
struct node
{
	int l,r;
}poi[2000100];
bool cmp(struct node a,struct node b)
{
	if(a.r!=b.r)
		return a.r<b.r;
	else
		return a.l>b.l;
}
int main()
{
	n=read();
	rep(i,1,n)
	{
		poi[i].l=read();
		poi[i].r=read();
	}
	sort(poi+1,poi+n+1,cmp);
	int las=-1;
	rep(i,1,n)
	{
		if(poi[i].l>=las)
		{
			ans++;
			las=poi[i].r;
		}
	}
	printf("%d",ans);
	return 0;
}
```
