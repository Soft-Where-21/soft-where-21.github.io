# 题意

题目中给出了n个子区间，求互不相交的子区间的最大个数。



# 题解

每个节目都有开始时间和结束时间，为了更多地收看节目，则转化为数学问题就是更多地选择不相交的区间。

那么我们可以首先将将n个节目按结束时间非减序排列，这里我用到的是归并排序，然后从第一个节目开始，依次遍历，若当前的节目的开始时间大于等于上一个节目的结束时间时，则将此节目添加到要看的节目当中，并更新最后节目的结束时间。

由于排序之后第一个节目是必选的，所以我们可以直接从i=1开始遍历。




# 代码

```cpp

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int s;//开始时间
    int e;//结束时间

}tv[2000000],tv2[2000000];
void merge(int low,int mid,int high)
{
    int i,j,k;
    for(k=low;k<=high;k++)
    {
        tv2[k]=tv[k];
    }
    i=low;
    j=mid+1;
    k=i;
    for(;i<=mid&&j<=high;k++)
    {
        if(tv2[i].e<=tv2[j].e)
        {
            tv[k]=tv2[i];
            i++;
        }
        else
        {
            tv[k]=tv2[j];
            j++;

        }

    }
    while(i<=mid)
    {
        tv[k]=tv2[i];
        k++;
        i++;
    }
    while(j<=high)
    {
        tv[k]=tv2[j];
        k++;
        j++;
    }
}
void merge2(int low,int high)
{
    int temp;
    if(low<high)
    {
        temp=(low+high)/2;
        merge2(low,temp);
        merge2(temp+1,high);
        merge(low,temp,high);
    }
}
int main()
{
    int n,i,j,temp;
    int out=1;
    scanf("%d",&n);
    for(i=0;i<n;i++)
    {
        scanf("%d%d",&tv[i].s,&tv[i].e);
    }
    merge2(0,n-1);//按结束时间来归并排序
    temp=tv[0].e;
    for(i=1;i<n;i++)
    {
        if(tv[i].s>=temp)
        {
            out++;
            temp=tv[i].e;
        }
    }
    printf("%d",out);
    return 0;
}
```
