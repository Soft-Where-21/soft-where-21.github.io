o(n)

## 参考代码

```c++
#include<iostream>
#include<algorithm>
using namespace std;
int arr[4][1010];

int main(){
    int n, a, b, c, d;
    cin >> n >> a >> b >> c >> d;
    for(int i = 0; i < 4; i++){
        for(int j = 1; j <= n; j++)
            cin >> arr[i][j];
    }
    int min = 100000000, minn=-1;
    for(int j = 1; j <= n; j++) {
        int aaa = abs(a-arr[0][j]) +abs(b-arr[1][j]) +abs(c-arr[2][j]) +abs(d-arr[3][j]);
        if(aaa < min){
            min = aaa;
            minn = j;
        }
    }
    cout << minn;

    return 0;
}
```
