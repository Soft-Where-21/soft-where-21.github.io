## 题目分析

​	在一个数组中寻找加和最大的连续片断，首先考虑到可以使用前缀和做法，先预处理每一位的前缀和，再枚举起点与终点，每次比较答案与终点前缀和减去起点前一个点的前缀和，存下较大的一个，此种做法时间复杂度为$O(n^2)$，不符合题目数据要求，然后考虑有没有线性时间复杂度做法。

​	最终输出答案记录为$ans$,中间记录的数组子段和记为$sum$，从数组起点开始扫描，则$sum=max(sum+a[i],a[i])$，前一种情况为延续之前的子段，后一种情况为从当前位置重新开始，每次比较一下$sum$与$ans$的大小，$ans$存储最大的$sum$，随后输出即可

## 代码



```c++
#include<queue>
#include<stdio.h>
#include<iostream>
#include<algorithm>
#include<cmath>
#include<map>
#include<string>
#include<string.h>
#include<ctype.h>
#include<stdlib.h>
#define rep(i,st,en) for(ll i=st;i<=en;i++)
typedef long long ll;
using namespace std;

inline int read()
{
    int x=0,f=1;
    char ch=getchar();
    while(ch<'0'||ch>'9'){if(ch=='-')f=-1;ch=getchar();}
    while(ch>='0'&&ch<='9'){x=x*10+ch-'0';ch=getchar();}
    return x*f;
}
ll n,a[1000100],ans=-999999999999,sum;
int main()
{
	
	scanf("%lld",&n);
	rep(i,1ll,n)
		scanf("%lld",&a[i]);
	rep(i,1ll,n)
	{
		sum=max(a[i],sum+a[i]);
		ans=max(ans,sum);
	}
	printf("%lld",ans);
	return 0;
}
```
