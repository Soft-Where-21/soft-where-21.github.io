# C2-F

## 题意

二分查找，即实现$lower\_bound()$和$upper\_bound()$

具体实现细节看代码即可。

$upper\_bound()$可以在数组末尾加一个很大的数，用来处理一些特殊情况。这里使用的是特判。

## Code

```c++
#include<stdio.h>
#include<string.h>
#include<math.h>
#define MN (100000+5)
int n,m;
int a[MN];
void find1(int x){
	int l=1,r=n;
	if(x>a[n]||x<a[1]){puts("Not Found");return;}
	while(l<r){
		int mid=l+r>>1;
		if(a[mid]>=x)r=mid;
		else l=mid+1;
	}
	if(a[r]==x)printf("%d\n",r);
	else puts("Not Found");
}
void find2(int x){
	int l=1,r=n;
	if(x>a[n]||x<a[1]){puts("Not Found");return;}
	while(l<r){
		int mid=l+r>>1;
		if(a[mid]>x)r=mid;
		else l=mid+1;
	}
	if(a[l]==x)printf("%d\n",l);
	else if(a[l-1]==x)printf("%d\n",l-1);
	else puts("Not Found");	
}
int main(){
	#ifndef ONLINE_JUDGE
	freopen("a.in","r",stdin);
	#endif
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;++i)scanf("%d",&a[i]);
	for(int i=1;i<=m;++i){
		int x,op;
		scanf("%d%d",&op,&x);
		if(op==1)
			find1(x);
		else find2(x);
	}
	return 0;
}
```
