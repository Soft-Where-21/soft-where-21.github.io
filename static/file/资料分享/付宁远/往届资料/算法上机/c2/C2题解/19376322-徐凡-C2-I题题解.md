## 题目分析

求平面最近点对的变形。（指从求欧氏距离变为求曼哈顿距离）首先按照x从小到大排序，利用归并的方法，如果左端点与右端点序号相邻，则返回它俩的距离，否则二分往下记录最小距离，记录下来后暴力从左端点到右端点跑一遍，将符合与中点距离小于等于记录下的最小距离的点扔进临时数组，把临时数组按照y从小到大排序，最后在临时数组中暴力求解记录答案即可

## 代码

```c++
#include<stdio.h>
#include<iostream>
#include<algorithm>
#include<cmath>
#include<map>
#include<string>
#include<queue>
#include<vector>
#include<string.h>
#include<ctype.h>
#include<stdlib.h>
#define rep(i,st,en) for(int i=st;i<=en;i++)
typedef long long ll;
using namespace std;
int n,t;
struct node
{
	double x;
	double y;
}poi[300010],tmp[300010];
bool cmp1(struct node a,struct node b)
{
	if(a.x!=b.x)
		return a.x<b.x;
	else
		return a.y<b.y;
}
bool cmp2(struct node a,struct node b)
{
	if(a.y!=b.y)
		return a.y<b.y;
	else
		return a.x<b.x;
}
double dis(struct node a,struct node b)
{
	return abs(a.x-b.x)+abs(a.y-b.y);
}
double mer(int l,int r)
{
	if(l>=r)
		return 999999999999999.9;
	if(l+1==r)
		return dis(poi[l],poi[r]);
	int mid=(l+r)>>1;
	double ans=min(mer(l,mid),mer(mid+1,r));
	int cnt=0;
	rep(i,l,r)
		if(fabs(poi[i].x-poi[mid].x)<ans)
			tmp[cnt++]=poi[i];
	sort(tmp,tmp+cnt,cmp2);
	rep(i,0,cnt-1)
	{
		for(int j=i+1;j<cnt&&fabs(tmp[i].y-tmp[j].y)<=ans;j++)
		{
			ans=min(ans,dis(tmp[i],tmp[j]));
		}
	}
	return ans;
}
int main()
{
	
	scanf("%d",&n);
	rep(i,0,n-1)
		scanf("%lf%lf",&poi[i].x,&poi[i].y);
	sort(poi,poi+n,cmp1);
	printf("%.3lf",mer(0,n-1));
	return 0;
}
```
