# 题解
   由题意可以考虑到从左向右读入数据的同时，通过**栈**来实现配对。考虑以下三种情况：
 
 - 当读入数字比上一数字小时，将数字和坐标压栈。
 
 - 当读入数字和上一数字相等时，更新栈顶元素的坐标。
 
 - 当读入数字比上一数字大时，将当前读入数字作为右侧，将上一元素（栈顶）出栈，并作为装水的底部，读取新的栈顶作为装水的左侧（由上知此时栈顶一定更高），并计算装水量，若左侧比右侧低则重复 **出栈作底，栈顶作左** ，直到左侧不低于右侧或者栈为空。
 
 > 可以使用快读来提速。
 ***
   ### 复杂度分析：
   
   由于若进入`while`循环，每执行一次则会使得栈内元素减少1，而栈内元素至多有`n`个，故时间复杂度为`O(n)`
   
   由**栈**知空间复杂度也为`O(n)`
   
 ***

# 代码
```cpp
#include <cstdio>
#include <iostream>
#include <vector>

#define LL long long

using namespace std;

inline int read() {
    int x = 0, f = 1;
    char ch = getchar();
    while(ch < '0' || ch > '9')
        ch = getchar();
    while(ch >= '0' && ch <= '9') {
        x = (x<<1)+(x<<3)+(ch^48);
        ch = getchar();
    }
    return x * f;
}

int main() {
    LL ans = 0;
    vector<pair<int, int>> a;
    int n, num, bottom, lastnum = 0x3f3f3f3f; // 用一个较大值表示空栈
    cin >> n;
    for (int i = 0; i < n; ++i) {
        num = read();
        while (num > lastnum) { // 比栈顶元素大
            bottom = a[a.size()-1].first;
            a.pop_back();
            if (!a.empty()) { // 若栈非空，往回找比弹出元素大的，并计算
                ans += (i - a[a.size()-1].second - 1) * (LL)(min(a[a.size()-1].first, num) - bottom);
                lastnum = a[a.size()-1].first; // 更新栈顶值
            }
            else { // 栈空了，则停止循环并入栈
                lastnum = 0x3f3f3f3f;
                break;
            }
        }
        if (num < lastnum) // 小于或者空栈时入栈
            a.push_back(pair<int, int>(num, i));
        if (num == lastnum) // 更新栈顶坐标
            a[a.size()-1].second = i;
        else // 更新值
            lastnum = num;
    }
    cout << ans;
    return 0;
}
```
